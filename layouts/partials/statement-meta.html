{{- /* statement-meta.html — resolve one statement's kind and section-scoped label. Takes a dict of "page" and "id" and returns a dict of kind, label, and title. The kind is "" when that page declares no such statement, and the label is "" for an unnumbered remark. Values come from the cached math-registry pass. */ -}} {{- $page := .page -}} {{- $registry := partialCached "math-registry.html" $page $page.RelPermalink -}} {{- $entry := index $registry.statements .id -}} {{- if not $entry -}} {{- $entry = dict "id" .id "kind" "" "label" "" "title" "" -}} {{- end -}} {{- return $entry -}}
