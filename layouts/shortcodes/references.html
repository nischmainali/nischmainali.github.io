{{- /* Render only the works cited earlier on this page. Author-year citations use an alphabetical bibliography even though Page.Store records first use. This is semantic static HTML: citation occurrence backlinks remain complete when scripts, styles, or the environmental renderer are unavailable. */ -}} {{- $page := .Page -}} {{- $registry := hugo.Data.citations -}} {{- $position := .Position -}} {{- if gt (len .Params) 0 -}} {{- errorf "references at %s does not accept arguments" $position -}} {{- end -}} {{- $order := default (slice) ($page.Store.Get "citationOrder") -}} {{- $occurrences := default (slice) ($page.Store.Get "citationOccurrences") -}} {{- if eq (len $order) 0 -}} {{- errorf "references at %s requires at least one earlier cite shortcode" $position -}} {{- end -}} {{- $entries := slice -}} {{- range $order -}} {{- $key := . -}} {{- $entry := index $registry $key -}} {{- $entries = $entries | append (dict "key" $key "sort" (lower (printf "%s %s %s" $entry.authors $entry.year $entry.title))) -}} {{- end -}}
    {{- range sort $entries "sort" -}} {{- $key := .key -}} {{- $entry := index $registry $key -}} {{- $entryOccurrences := slice -}} {{- range $occurrences -}} {{- if eq .key $key -}} {{- $entryOccurrences = $entryOccurrences | append . -}} {{- end -}} {{- end -}} {{- /* Ordinals are document order, so the backlinks number the way the reader met them. */ -}} {{- $entryOccurrences = sort $entryOccurrences "ordinal" -}} {{- $occurrenceCount := len $entryOccurrences -}}
  1. {{ $entry.authors }} ({{ $entry.year }}). {{ $entry.title }}. {{ $entry.publication }}. {{ $entry.identifier }}. {{- range $index, $occurrence := $entryOccurrences -}} {{- if gt $index 0 }} {{ end -}} ↩{{ if gt $occurrenceCount 1 }}{{ add $index 1 }}{{ end }} {{- end -}}
    2. {{- end -}}
