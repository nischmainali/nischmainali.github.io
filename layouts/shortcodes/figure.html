{{- /* Editorial figure interface. Required: src, and alt unless decorative="true" Optional: placement="measure|wide|margin" (wide="true" remains supported) treatment="natural|ink" priority="true", sizes, link, width, height, loading, class, title, caption, attr, and attrlink */ -}} {{- $src := strings.TrimSpace (default "" (.Get "src")) -}} {{- $alt := strings.TrimSpace (default "" (.Get "alt")) -}} {{- $decorative := eq (lower (default "false" (.Get "decorative"))) "true" -}} {{- $priority := eq (lower (default "false" (.Get "priority"))) "true" -}} {{- $legacyWide := eq (lower (default "false" (.Get "wide"))) "true" -}} {{- $zoomValue := lower (default "true" (.Get "zoom")) -}} {{- if not (in (slice "true" "false") $zoomValue) -}} {{- errorf "figure at %s has invalid zoom value %q; use true or false" .Position $zoomValue -}} {{- end -}} {{- $zoom := eq $zoomValue "true" -}} {{- $placement := lower (default "measure" (.Get "placement")) -}} {{- if and $legacyWide (not (.Get "placement")) -}} {{- $placement = "wide" -}} {{- end -}} {{- if not (in (slice "measure" "wide" "margin") $placement) -}} {{- errorf "figure at %s has invalid placement %q; use measure, wide, or margin" .Position $placement -}} {{- end -}} {{- if eq $placement "margin" -}} {{- .Page.Store.Set "articleHasMarginalia" true -}} {{- end -}} {{- if and (eq $placement "margin") (not (strings.TrimSpace (default "" (.Get "caption")))) -}} {{- errorf "margin figure at %s requires a caption" .Position -}} {{- end -}} {{- $treatment := lower (default "natural" (.Get "treatment")) -}} {{- if not (in (slice "natural" "ink") $treatment) -}} {{- errorf "figure at %s has invalid treatment %q; use natural or ink" .Position $treatment -}} {{- end -}} {{- $link := strings.TrimSpace (default "" (.Get "link")) -}} {{- if and $decorative $link -}} {{- errorf "figure at %s cannot be both decorative and a link; linked images need meaningful alt text" .Position -}} {{- end -}} {{- $loading := lower (default "lazy" (.Get "loading")) -}} {{- if $priority -}} {{- $loading = "eager" -}} {{- end -}} {{- if not (in (slice "lazy" "eager") $loading) -}} {{- errorf "figure at %s has invalid loading value %q; use lazy or eager" .Position $loading -}} {{- end -}}
