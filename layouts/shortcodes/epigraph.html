{{- $quote := strings.TrimSpace .Inner -}} {{- $attribution := strings.TrimSpace (default "" (.Get "attribution")) -}} {{- $source := strings.TrimSpace (default "" (.Get "source")) -}} {{- $link := strings.TrimSpace (default "" (.Get "link")) -}} {{- if not $quote -}} {{- errorf "epigraph at %s requires a quotation" .Position -}} {{- end -}} {{- if and (not $attribution) (not $source) -}} {{- errorf "epigraph at %s requires attribution or source" .Position -}} {{- end -}} {{- if and $link (not (or $attribution $source)) -}} {{- errorf "epigraph at %s cannot use link without attribution or source" .Position -}} {{- end -}}
